2. Integrasi Google Sheets API

Kosongkan jika hanya ingin menyimpan data di browser (Local Storage).

Petunjuk Cara Setup Google Apps Script
  1. Buat **Google Spreadsheet** baru di Google Drive Anda.
  2. Buka menu **Ekstensi > Apps Script**.
  3. Hapus semua kode bawaan, lalu **copypaste script berikut**: 
    function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
  var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
  
  var members = sheetM.getDataRange().getValues();
  var history = sheetH.getDataRange().getValues();
  
  return ContentService.createTextOutput(JSON.stringify({
    members: members.length > 1 ? members.slice(1) : [],
    history: history.length > 1 ? history.slice(1) : []
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if(data.action === "syncAll") {
    // Sync Members
    var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
    sheetM.clear();
    sheetM.appendRow(["ID", "Nama", "Kelas"]);
    data.members.forEach(function(m) { sheetM.appendRow([m.id, m.name, m.class]); });
    
    // Sync History
    var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
    sheetH.clear();
    sheetH.appendRow(["Tanggal", "Waktu", "ID", "Nama", "Kelas", "Status"]);
    data.history.forEach(function(h) { sheetH.appendRow([h.date, h.time, h.memberId, h.memberName, h.memberClass, h.status]); });
  }
  return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
}
  4. Klik **Terapkan (Deploy) > Terapkan Sebagai Aplikasi Web**.
  5. Atur **Akses Siapa Saja (Anyone)**, lalu klik Deploy.
  6. Salin **URL Web App** dan tempelkan pada kolom di atas.
