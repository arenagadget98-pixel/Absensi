Pengaturan Presensi & Notifikasi

Konfigurasi batas waktu presensi, integrasi Telegram Bot, suara ucapan Bahasa Indonesia, dan Google Spreadsheet API. Seluruh pengaturan tersimpan secara otomatis di Google Spreadsheet.

1. Batas Waktu Presensi

Presensi setelah jam ini otomatis dicatat **Terlambat**.

2. Modus Suara & Ucapan Bahasa Indonesia (Web Speech API)

Saat Presensi berhasil, sistem menyapa: "Selamat datang, [Nama]. Presensi [Status]" menggunakan Web Speech API Bahasa Indonesia (id-ID).

3. Integrasi Telegram Notification

Notifikasi Otomatis Telegram Kirim pesan otomatis ke Telegram setiap ada anggota presensi.

Dapatkan Bot Token dari @BotFather dan Chat ID dari @userinfobot.

4. Integrasi Google Spreadsheet API (Cloud Store)

Petunjuk Deploy Google Apps Script (Mendukung Simpan Pengaturan)
  1. Buat **Google Spreadsheet** baru di Google Drive Anda.
  2. Buka menu **Ekstensi > Apps Script**.
  3. Hapus semua kode bawaan, lalu **copypaste script Google Apps Script terbaru di bawah ini**: 
    function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
  var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
  var sheetS = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
  
  var membersData = sheetM.getDataRange().getValues();
  var historyData = sheetH.getDataRange().getValues();
  var settingsData = sheetS.getDataRange().getValues();
  
  var members = [];
  if (membersData.length > 1) {
    for (var i = 1; i < membersData.length; i++) {
      if (membersData[i][0]) {
        members.push({ 
          id: String(membersData[i][0]), 
          name: String(membersData[i][1]), 
          class: String(membersData[i][2]),
          nfcId: String(membersData[i][3] || "")
        });
      }
    }
  }

  var history = [];
  if (historyData.length > 1) {
    for (var j = 1; j < historyData.length; j++) {
      if (historyData[j][0]) {
        history.push({
          date: String(historyData[j][0]),
          time: String(historyData[j][1]),
          memberId: String(historyData[j][2]),
          memberName: String(historyData[j][3]),
          memberClass: String(historyData[j][4]),
          status: String(historyData[j][5])
        });
      }
    }
  }

  var settings = {};
  if (settingsData.length > 1) {
    for (var k = 1; k < settingsData.length; k++) {
      if (settingsData[k][0]) {
        var val = settingsData[k][1];
        if (val instanceof Date) {
          var h = ("0" + val.getHours()).slice(-2);
          var m = ("0" + val.getMinutes()).slice(-2);
          val = h + ":" + m;
        } else {
          val = String(val);
        }
        settings[String(settingsData[k][0])] = val;
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    members: members,
    history: history,
    settings: settings
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if(data.action === "syncAll") {
    // 1. Sync Data Anggota
    var sheetM = ss.getSheetByName("Anggota") || ss.insertSheet("Anggota");
    sheetM.clear();
    sheetM.appendRow(["ID", "Nama", "Kelas", "No NFC"]);
    if(data.members && data.members.length > 0) {
      data.members.forEach(function(m) {
        sheetM.appendRow(["'" + m.id, m.name, m.class, "'" + (m.nfcId || "")]);
      });
    }
    
    // 2. Sync Riwayat Absensi
    var sheetH = ss.getSheetByName("Riwayat") || ss.insertSheet("Riwayat");
    sheetH.clear();
    sheetH.appendRow(["Tanggal", "Waktu", "ID", "Nama", "Kelas", "Status"]);
    if(data.history && data.history.length > 0) {
      data.history.forEach(function(h) {
        sheetH.appendRow(["'" + h.date, "'" + h.time, "'" + h.memberId, h.memberName, h.memberClass, h.status]);
      });
    }

    // 3. Sync Pengaturan & Notifikasi
    if(data.settings) {
      var sheetS = ss.getSheetByName("Pengaturan") || ss.insertSheet("Pengaturan");
      sheetS.clear();
      sheetS.appendRow(["Pengaturan", "Nilai"]);
      for (var key in data.settings) {
        sheetS.appendRow([key, "'" + String(data.settings[key])]);
      }
    }
  }
  return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
}
